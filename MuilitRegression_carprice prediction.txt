import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_absolute_error
from sklearn.metrics import mean_squared_error
df=pd.read_csv("carprice.csv")
print(df)
x=df[["Year","Kms_Driven","Age"]]
y=df["Price"]
model=LinearRegression()
model.fit(x,y)
predicted= model.predict([[2025,40000,2]])
print("predicted price:",predicted)
print("Intercept:",model.intercept_)
print("Coefficients:",model.coef_)
print("R2 score:",model.score(x,y))
predicted= model.predict(x)
print("Actual value:",y.values)
print("predicted value:",predicted)
mae=mean_absolute_error(y,predicted)
print("MAE:",mae)
mse=mean_squared_error(y,predicted)
print("MSE:",mse)
plt.figure(figsize=(8,5))
plt.scatter(y,predicted,color="red")
plt.plot(range(len(y)), y, marker="o", label="Actual Price")
plt.plot(range(len(predicted)), predicted, marker="o", label="Predicted Price")
plt.xlabel("Actual price")
plt.ylabel("Predicted price")
plt.title("Car_price")
plt.show()
plt.figure(figsize=(8,5))
plt.bar(range(len(y)),y,width=0.4,label="Actual price")
plt.bar([i+ 0.4 for i in range(len(y))],predicted, width=0.4,label="Predicted price")
plt.xlabel("Actual price")
plt.ylabel("Predicted price")
plt.title("Car_actualprice")
plt.legend()
plt.show()
